cmake_minimum_required(VERSION 3.18)
project(loa_allocator_smoke LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Threads REQUIRED)
include(${LOA_SOURCE_DIR}/cmake/options.cmake)
include(${LOA_SOURCE_DIR}/cmake/allocator.cmake)
add_executable(allocator_smoke main.cpp)
target_link_libraries(allocator_smoke PRIVATE server_allocator Threads::Threads)
if(allocator_selected STREQUAL "MIMALLOC")
  target_compile_definitions(allocator_smoke PRIVATE LOA_TEST_MIMALLOC)
endif()
enable_testing()
add_test(NAME allocator_smoke COMMAND allocator_smoke)
